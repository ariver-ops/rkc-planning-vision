// ---------- Recalcul indépendant des horaires, des pauses et des créneaux ----------
// Ce dossier ne reprend pas ces calculs au moteur : il les refait à partir des données, puis compare.
const minutes = (t) => { const [a, b] = t.split(':').map(Number); return a * 60 + b; };
const intervalle = (de, a) => { const x = minutes(de); let y = minutes(a); if (y <= x) y += 1440; return [x, y]; };
function horairesActivite(a, cle) {
  const hs = a.horaires || {};
  const x = hs[cle] || (jourDe(cle).weekend ? hs.weekend : hs.semaine) || [a.debut, a.fin];
  return intervalle(x[0], x[1]);
}
function horairesPoste(p, cle, acts) {
  const propre = p.horaires && (p.horaires[cle] || (jourDe(cle).weekend ? p.horaires.weekend : p.horaires.semaine));
  if (propre) return intervalle(propre[0], propre[1]);
  if (p.debut) return intervalle(p.debut, p.fin);
  return horairesActivite(activite(p.activite, acts), cle);
}
// Pause comprise dans l'amplitude : 1 h pour une journée de 9 heures ou plus, 30 min au-delà de 6 heures

const travail = (pl) => pl.fin - pl.debut - pauseAttendue(posteDe(pl.poste), pl.fin - pl.debut);

// La journée d'un extra, recalculée ici : de la première prise de poste à la dernière fin de poste des activités du jour
// hors événements, pour les rôles qu'il tient (l'accueil, la piste, l'atelier, le magasin ont chacun leurs horaires)
function journeeDe(cle, p, acts) {
  let de = Infinity, fin = -Infinity;
  for (const po of d.postes) {
    const a = activite(po.activite, acts);
    if (!a || estEvenementDossier(a) || !joursActivite(a).includes(cle) || (po.jours && !po.jours.includes(cle)) || !nombreDe(po, cle)) continue;
    if (p && !p.metiers[po.metier]) continue;
    const [x, y] = horairesPoste(po, cle, acts);
    de = Math.min(de, x); fin = Math.max(fin, y);
  }
  return de < fin ? [de, fin] : null;
}
// Ce qu'un extra a coché un jour donné : la journée (aux horaires de son rôle), chaque événement (ses horaires ce jour-là) ;
// le moteur lit aussi « toute » et un créneau « de-à ». Pour une place donnée, seul compte ce qui vaut pour elle : la
// journée pour un poste de la journée, l'événement lui-même pour un poste d'événement. Sans place, chaque coche est listée
// à part ; pour une place, les coches qui valent et se touchent n'en font qu'une.
function creneauxDonnes(cle, dispo, acts, p, place) {
  if (!dispo) return [];
  const actPlace = place ? activiteDePlace(place, acts) : null;
  const evtPlace = place ? estEvenementDossier(actPlace) : null;
  const vaut = (s) => {
    if (!place || s === 'toute') return true;
    if (s === 'journee') return !evtPlace;
    const a = activite(s, acts);
    if (a) return estEvenementDossier(a) ? !!actPlace && s === actPlace.id : !evtPlace;
    return true;
  };
  const items = (dispo === 'toute' ? ['toute'] : dispo).filter(vaut).map((s) => {
    if (s === 'toute') return { de: 0, fin: 2880, etiquettes: [], toute: true };
    if (s === 'journee') { const jr = journeeDe(cle, p, acts); return jr && { de: jr[0], fin: jr[1], etiquettes: ['la journée'], journee: true }; }
    const a = activite(s, acts);
    if (a) { const [de, fin] = horairesActivite(a, cle); return { de, fin, etiquettes: [nomActivite(a)], activites: [a.id] }; }
    const [x, y] = s.split('-');
    const [de, fin] = intervalle(x, y);
    return { de, fin, etiquettes: [], libre: true };
  }).filter(Boolean).sort((u, v) => u.de - v.de);
  if (!place) return items.map((c) => Object.assign({}, c, { activites: (c.activites || []).slice() }));
  const res = [];
  for (const c of items) {
    const der = res[res.length - 1];
    if (der && c.de <= der.fin) {
      der.fin = Math.max(der.fin, c.fin);
      der.etiquettes.push(...c.etiquettes);
      der.activites.push(...(c.activites || []));
      der.toute = der.toute || c.toute;
      der.libre = der.libre || c.libre;
      der.journee = der.journee || c.journee;
    } else res.push(Object.assign({}, c, { etiquettes: c.etiquettes.slice(), activites: (c.activites || []).slice() }));
  }
  return res;
}

  return (noms.length ? et(noms) + ', ' : '') + plage(c.de, c.fin);

// La journée et la soirée enchaînées, recalculées ici : le soir, le RKC prend d'abord ceux qui sont déjà là. Une place de
// la journée puis un événement qui commence à sa fin, ou au plus une heure après, le même jour : une seule présence, une
// seule pause comptée sur toute l'amplitude (l'heure creuse entre les deux en tient lieu).
const ENCHAINEMENT = 60;
const sEnchainent = (a, b, acts, ajouts) => !estEvenementDossier(activiteDePlace(a, acts, ajouts)) && estEvenementDossier(activiteDePlace(b, acts, ajouts)) && b.debut >= a.fin && b.debut - a.fin <= ENCHAINEMENT;
function presencesDossier(ms, acts, ajouts) {
  const res = [];
  for (const j of JOURS) {
    const du = ms.filter((m) => m.j === j).sort((a, b) => a.pl.debut - b.pl.debut);
    if (!du.length) continue;
    const blocs = du.length === 2 && sEnchainent(du[0].pl, du[1].pl, acts, ajouts) ? [du] : du.map((m) => [m]);
    for (const b of blocs) {
      const i = b[0].i, de = b[0].pl.debut, fin = b[b.length - 1].pl.fin;
      const amplitude = fin - de;
      const t = b.length === 1 ? travail(b[0].pl) : amplitude - Math.max(pauseAttendue(null, amplitude), b[1].pl.debut - b[0].pl.fin);
      res.push({ i, j, debut: i * 1440 + de, fin: i * 1440 + fin, travail: t, enchaine: b.length > 1, places: b.map((m) => m.pl) });
    }
  }
  return res;
}

// Combien de combinaisons de jours de CDI : recompté ici, selon la règle (repos d'affilée, en semaine d'abord)
function optionsCDI(p) {
  const presents = JOURS.filter((j) => !absence(p, j));
  const n = Math.min(p.contrat.jours, presents.length);
  if (presents.length <= n) return 1;
  const k = presents.length - n;
  const fenetres = [];
  for (let s = 0; s + k <= JOURS.length; s++) {
    const repos = JOURS.slice(s, s + k);
    if (repos.every((j) => presents.includes(j))) fenetres.push(repos.some((j) => jourDe(j).weekend));
  }
  return fenetres.filter((w) => !w).length || fenetres.length || 1;
}
const nbCombinaisons = CDIS.reduce((n, p) => n * optionsCDI(p), 1);

// ---------- Les repères des plannings réels, relus dans la note de calibration ----------
const NOTE = path.join(RACINE, 'analyse', 'immersion', 'plannings-reels.md');

const lireNote = (re, quoi) => { const m = notePlate.match(re); if (!m) throw new Error(relatif(NOTE) + ' : ' + quoi + ' introuvable (la note a changé de forme)'); return m; };
function tableauNote(titre) {
  const i = note.indexOf(titre);
  if (i < 0) throw new Error(relatif(NOTE) + ' : « ' + titre + ' » introuvable');
  const lignes = note.slice(i).split('\n');
  const k = lignes.findIndex((l) => l.startsWith('|'));
  const res = [];
  for (let x = k; x < lignes.length && lignes[x].startsWith('|'); x++) {
    const c = lignes[x].replace(/^\||\|$/g, '').split('|').map((s) => s.trim());
    if (!c.every((s) => /^:?-+:?$/.test(s))) res.push(c);
  }
  return res;
}
const enNombre = (s) => Number(String(s).replace(',', '.'));
function lireRepere(s, quoi) {
  const m = String(s).match(/^([\d,]+) \(([\d,]+) à ([\d,]+), max ([\d,]+)\)$/);
  if (!m) throw new Error(relatif(NOTE) + ' : ' + quoi + ' illisible (« ' + s + ' »)');
  return { med: m[1], q1: m[2], q3: m[3], max: m[4] };
}

async function ouvrirChrome(dossier) {
  const exe = CHROMES.find((f) => fs.existsSync(f));
  if (!exe) throw new Error('Chrome introuvable sur cette machine');
  const profil = path.join(dossier, 'profil');
  const proc = spawn(exe, ['--headless=new', '--disable-gpu', '--no-first-run', '--no-default-browser-check', '--remote-debugging-port=0', '--user-data-dir=' + profil, 'about:blank'], { stdio: 'ignore' });
  const fermerProc = () => { try { proc.kill('SIGKILL'); } catch { /* déjà fermé */ } };
  try {
    let port = null;
    for (let i = 0; i < 200 && !port; i++) {
      try { port = fs.readFileSync(path.join(profil, 'DevToolsActivePort'), 'utf8').split('\n')[0].trim() || null; } catch { /* démarrage */ }
      if (!port) await attendre(100);
    }
    if (!port) throw new Error('Chrome ne répond pas');
    let cible = null;
    for (let i = 0; i < 100 && !cible; i++) {
      try { cible = (await (await fetch('http://127.0.0.1:' + port + '/json/list')).json()).find((t) => t.type === 'page'); } catch { /* démarrage */ }
      if (!cible) await attendre(100);
    }
    if (!cible) throw new Error('aucun onglet dans Chrome');
    const ws = new WebSocket(cible.webSocketDebuggerUrl);
    await new Promise((ok, ko) => { ws.onopen = ok; ws.onerror = () => ko(new Error('connexion à Chrome impossible')); });
    let n = 0;
    const attente = new Map(), ecouteurs = new Set(), erreurs = [];
    ws.onmessage = (ev) => {
      const m = JSON.parse(ev.data);
      if (m.id && attente.has(m.id)) { const { ok, ko } = attente.get(m.id); attente.delete(m.id); if (m.error) ko(new Error(m.error.message)); else ok(m.result); return; }
      if (m.method === 'Runtime.exceptionThrown') { const x = m.params.exceptionDetails; erreurs.push(x.exception ? x.exception.description : x.text); }
      if (m.method === 'Runtime.consoleAPICalled' && m.params.type === 'error') erreurs.push(m.params.args.map((a) => a.value || a.description).join(' '));
      if (m.method === 'Log.entryAdded' && m.params.entry.level === 'error' && !/fonts\.(googleapis|gstatic)\.com/.test(m.params.entry.url || '')) erreurs.push(m.params.entry.text);
      for (const f of ecouteurs) f(m);
    };
    const envoyer = (method, params) => new Promise((ok, ko) => { const id = ++n; attente.set(id, { ok, ko }); ws.send(JSON.stringify({ id, method, params: params || {} })); });
    await envoyer('Page.enable');
    await envoyer('Runtime.enable');
    await envoyer('Log.enable');
    // Les polices en ligne n'ont aucun effet sur le planning : on les bloque, pour une vérification rapide et sans réseau
    await envoyer('Network.enable');
    await envoyer('Network.setBlockedURLs', { urls: ['*fonts.googleapis.com*', '*fonts.gstatic.com*'] });
    return {
      erreurs,
      async ouvrir(url) {
        erreurs.length = 0;
        const charge = new Promise((ok) => { const f = (m) => { if (m.method === 'Page.loadEventFired') { ecouteurs.delete(f); ok(); } }; ecouteurs.add(f); });
        await envoyer('Page.navigate', { url });
        await Promise.race([charge, attendre(20000)]);
      },
      async evaluer(expression) {
        const r = await envoyer('Runtime.evaluate', { expression, returnByValue: true, awaitPromise: true });
        if (r.exceptionDetails) throw new Error(r.exceptionDetails.exception ? r.exceptionDetails.exception.description : r.exceptionDetails.text);
        return r.result.value;
      },
      fermer() { try { ws.close(); } catch { /* déjà fermé */ } fermerProc(); },
    };
  } catch (e) { fermerProc(); throw e; }
}

function calculerUrgences(dm, Pm, plan) {
  const parPersonne = (id) => dm.personnes.find((p) => p.id === id);
  return dm.imprevus.map((s) => {
    if (!s.ajouts) {
      const absent = parPersonne(s.absent);
      const r = Pm.remplacants(dm, plan, s.absent, s.jour);
      const choisi = r.candidats[0] || null;
      const apres = choisi ? Pm.remplacer(plan, s.jour, s.absent, choisi.id) : plan;
      const opts = absent.type === 'CDI' ? { imprevus: { [s.absent]: [s.jour] } } : {};
      const controle = Pm.controler(dm, apres, opts);
      const sansInfo = absent.type === 'CDI' ? Pm.controler(dm, apres) : null;
      return { s, type: 'remplacement', absent, r, choisi, controle, sansInfo };
    }
    const d2 = Object.assign({}, dm, { activites: dm.activites.concat([s.activite]) });
    const a = s.activite;
    const avant = Pm.renforts(d2, plan, s.ajouts, s.jour);
    const extrasPresents = dm.personnes.filter((p) => p.type === 'Extra' && !(p.absences || []).some((x) => x.jours.includes(s.jour)));
    const tousOui = JSON.parse(JSON.stringify(dm.reponses));
    for (const p of extrasPresents) { tousOui[p.id] = tousOui[p.id] || {}; const v = tousOui[p.id][s.jour]; tousOui[p.id][s.jour] = (v === 'toute' ? ['toute'] : (v || [])).concat([a.id]); }
    const cibles = new Set();
    for (const x of avant.places) {
      const r = Pm.remplacants(d2, plan, '', s.jour, { place: x.place, reponses: tousOui, sollicites: extrasPresents.map((p) => p.id) });
      for (const c of r.candidats) if (c.rang === 1 && parPersonne(c.id).type === 'Extra') cibles.add(c.id);
    }
    const oui = Object.entries(s.reponsesDemande).filter(([, v]) => v.reponse === 'oui').map(([id]) => id).filter((id) => cibles.has(id));
    const non = Object.entries(s.reponsesDemande).filter(([, v]) => v.reponse === 'non').map(([id]) => id).filter((id) => cibles.has(id));
    const horsCible = Object.keys(s.reponsesDemande).filter((id) => !cibles.has(id));
    const reponses = JSON.parse(JSON.stringify(dm.reponses));
    for (const id of oui) { reponses[id] = reponses[id] || {}; const v = reponses[id][s.jour]; reponses[id][s.jour] = (v === 'toute' ? ['toute'] : (v || [])).concat([a.id]); }
    const sollicites = [...new Set(dm.sollicites.concat(oui))];
    const apres = Pm.renforts(d2, plan, s.ajouts, s.jour, { reponses, sollicites });
    const controle = Pm.controler(d2, apres.plan, { ajouts: s.ajouts, reponses, sollicites });
    return { s, type: 'ajout', d2, avant, cibles: [...cibles], oui, non, horsCible, apres, controle };
  });
}

function fichierMesures() {
  if (argMesures) return path.resolve(argMesures);
  const dossier = path.join(RACINE, 'analyse');
  if (!fs.existsSync(dossier)) return null;
  const valides = fs.readdirSync(dossier).filter((f) => /^mesures-.*\.json$/.test(f)).map((f) => path.join(dossier, f)).filter((f) => {
    try { const x = JSON.parse(fs.readFileSync(f, 'utf8')); return Array.isArray(x) && x.length && x.every((y) => y.largeur && y.hauteur); } catch { return false; }
  });
  valides.sort((a, b) => fs.statSync(b).mtimeMs - fs.statSync(a).mtimeMs);
  return valides[0] || null;
}

    const ok = !!visee && !visee.ok;
    if (ok) attrapees++;
    exiger(ok, 'Erreur volontaire non signalée par la règle attendue : ' + texte);
    if (e.sixieme) {
      const six = c.regles.find((r) => r.regle.startsWith('Pas plus de 6 jours'));
      exiger(six.ok, 'Un sixième jour déclenche à tort la règle légale des 6 jours');
      msg += ' · règle légale des 6 jours : ' + (six.ok ? 'non déclenchée, à raison' : 'DÉCLENCHÉE À TORT');
    }
    ecrire(rangee(texte, visee ? visee.regle : e.regle, ok ? 'oui' : '**NON**', msg));
  }
  const couvertes = CTRL.regles.filter((r) => declenchees.has(r.regle)).length;
  const viseesTexte = new Set(erreurs.filter((e) => e.regle).map((e) => regleNom(e.regle)));
  const parRicochet = CTRL.regles.filter((r) => declenchees.has(r.regle) && !viseesTexte.has(r.regle)).map((r) => '« ' + r.regle.split(' :')[0] + ' »');
  const nbCreneaux = erreurs.filter((e) => e.creneau).length;
  ecrire('', '**Résultat : ' + attrapees + ' erreurs sur ' + nbErreurs + ' signalées par la bonne règle**, dont ' + nbCreneaux + ' où un extra est placé sur ce qu’il n’a pas coché ; le témoin ne déclenche rien. Chacune des ' + CTRL.regles.length + ' règles est déclenchée au moins une fois : ' + couvertes + ' sur ' + CTRL.regles.length + (parRicochet.length ? ' (' + et(parRicochet) + ' par ricochet, sans erreur qui lui soit propre)' : '') + '.', '');
  exiger(couvertes === CTRL.regles.length, 'Toutes les règles ne sont pas déclenchées par les erreurs volontaires');
  const iBref = L.findIndex((x) => x.startsWith('- Le contrôle **attrape les erreurs**'));
  if (iBref > -1) L[iBref] = '- Le contrôle **attrape les erreurs** : ' + attrapees + ' erreurs introduites volontairement sur ' + nbErreurs + ' sont signalées par la bonne règle, dont ' + nbCreneaux + ' où un extra est placé sur ce qu’il n’a pas coché ; un échange permis, servant de témoin, ne déclenche rien (partie 7.3).';

  // 7.4 Un changement à la main
  const lundiLea = !(d.joursCDI.lea || []).includes('lun') && !absence(personne('lea'), 'lun');
  exiger(lundiLea, 'Le changement à la main suppose Léa D. en repos le lundi');
  const manuel = clone();
  remplacerPar(manuel, 'lun', 'jade', 'lea');
  const cm = P.controler(d, manuel);
  exiger(!cm.ok, 'Le changement à la main n’est pas signalé');
  ecrire('### 7.4 Un changement à la main est recontrôlé', '',
    'Dans la page, le responsable relit la proposition et peut « Changer une case » avant de valider ; dans la maquette, ce bouton est une image. Le changement est donc fait ici, par le moteur : ' + nom('lea') + ', CDI en repos le lundi, est mise à l’accueil à la place de ' + nom('jade') + '. Le contrôle repassé répond : ' + cm.regles.filter((r) => !r.ok).map((r) => '« ' + r.regle + ' » non respectée (' + r.erreurs.join(' ; ') + ')').join(' ; ') + '. Le changement reste possible, mais il est signalé avant l’envoi.', '');

  // ======================================================================
  // 8. Les mesures de la page
  // ======================================================================
  ecrire('## 8. Ce qui a été mesuré sur la page', '');
  const fM = fichierMesures();
  const fichiersPage = [DIST, path.join(PAGE, 'index.html'), path.join(PAGE, 'app.js'), path.join(PAGE, 'styles.css'), path.join(ici, 'donnees.js'), path.join(ici, 'planning.js')].filter((f) => fs.existsSync(f));
  const pageModifiee = Math.max(...fichiersPage.map((f) => fs.statSync(f).mtimeMs));
  let hauteur390 = null;
  if (fM && fs.existsSync(fM)) {
    const m = JSON.parse(fs.readFileSync(fM, 'utf8'));
    const tM = fs.statSync(fM).mtimeMs;
    ecrire('Mesures faites dans Chrome sans fenêtre, sur la page construite, à ' + pluriel(m.length, 'largeur', 'largeurs') + ' d’écran (fichier ' + montrer(fM) + ', du ' + quand(tM) + ').', '',
      ...tete('Largeur', 'Hauteur de page', 'Débordement horizontal', 'Textes sous 12 px', 'Erreurs dans la console', 'Mots rejetés', 'Tirets longs'));
    for (const x of m) ecrire(rangee(x.largeur + ' px', x.hauteur + ' px', x.debordement ? '**oui**' : 'aucun', x.nbTextesSous12px, (x.erreursConsole || []).length, (x.motsRejetes || []).length, (x.tiretsLongs || []).length));
    const a390 = m.find((x) => x.largeur === 390);
    if (a390) hauteur390 = a390.hauteur;
    if (tM >= pageModifiee) ecrire('', 'Ces mesures sont postérieures à la dernière modification de la page (' + quand(pageModifiee) + ') : elles valent pour cette version.', '');
    else { ecrire('', '**Ces mesures datent d’avant la dernière modification de la page (' + quand(pageModifiee) + ') : elles sont à refaire.**', ''); avertissements.push('Mesures de page antérieures à la dernière modification de la page (' + fM + ')'); }
  } else { ecrire('Aucune mesure de page trouvée (fichier `analyse/mesures-*.json`).', ''); avertissements.push('Aucune mesure de page trouvée'); }
  if (C && C.defaut.temps) ecrire('Temps de calcul mesuré dans Chrome, dans la page : ' + C.defaut.temps.fige.toFixed(1).replace('.', ',') + ' ms pour la proposition affichée au chargement, ' + Math.round(C.defaut.temps.complet).toLocaleString('fr-FR') + ' ms pour un calcul complet de toutes les combinaisons de jours des CDI, que la page n’a pas besoin de faire.', '');

  // ======================================================================
  // 9. Limites
  // ======================================================================
  ecrire('## 9. Limites connues, dites franchement', '',
    '- Les personnes, leurs contrats et leurs réponses sont fictifs ; la semaine d’exemple est inventée sur le rythme des plannings réels. Les besoins par activité, les horaires, les seuils des groupes et les plafonds de pondération sont à paramétrer avec le RKC.',
    ecartsReels.length ? '- Écarts avec les plannings réels (partie 1.1) : ' + ecartsReels.join(' ; ') + '. La location de semaine compte ' + locSemaine + ' personnes en piste, contre ' + RA.LOC.hauteSemaine + ' dans les plannings réels.' : '- La semaine d’exemple reste dans les fourchettes des plannings réels (partie 1.1).',
    '- Dans les plannings réels, un extra enchaîne souvent la location l’après-midi et un groupe le soir. La maquette suit la règle « chacun garde le même poste toute la journée » : une personne, un poste par jour.',
    (() => {
      const [de6, fin6] = horairesActivite(course, jourCourse);
      const ampReelle = minutes(REEL.horaires['6HEURES'].amplitude.replace(/ h ?/, ':').replace(/:$/, ':00'));
      const travailReel = ampReelle - pauseAttendue(null, ampReelle);
      return '- Les ' + course.libelle + ' durent ici de ' + plage(de6, fin6) + ' (' + enHeures(fin6 - de6) + ' d’amplitude), moins que dans les plannings réels : le plus souvent de ' + REEL.horaires['6HEURES'].creneau + ', ' + REEL.horaires['6HEURES'].amplitude + ' d’amplitude en médiane. Avec la pause, ' + REEL.horaires['6HEURES'].amplitude + ' d’amplitude font ' + enHeures(travailReel) + ' de travail' + (travailReel > 10 * 60 ? ', plus que les 10 heures du minimum légal général : à voir avec le RKC (dérogation, ou deux équipes).' : '.');
    })(),
    '- Le repos de 35 heures se mesure dans la semaine seule, à partir du lundi 0 h, sans la fin de la semaine précédente. Le moteur remplit la semaine jour par jour : un changement un jour peut créer un trou plus tard (partie 7.1), toujours signalé.',
    '- L’envoi des SMS, l’arrivée des réponses et la demande ciblée sont simulés. Aucune intégration avec un logiciel de réservation ou d’envoi n’existe.',
    '- Les jours des CDI sont mémorisés pour que la page s’affiche aussitôt ; la page ne les recalcule pas elle-même. Ce dossier vérifie à chaque génération que le calcul complet retombe sur les mêmes jours (partie 3.1).',
    '- Sur téléphone, le planning se lit un jour à la fois (onglets).' + (hauteur390 ? ' La page reste longue : environ ' + Math.round(hauteur390 / 844) + ' écrans de haut à 390 pixels de large.' : ''), '');

  // ======================================================================
  // Contrôle du dossier lui-même : vocabulaire et typographie
  // ======================================================================
  const texte = L.join('\n');
  const interdits = [/[–—]/, /arbitr/i, /ajust/i, /mission/i, /relanc/i, /synchronis/i, /sans rien cacher/i, /actualisé en continu/i, /pointage/i];
  for (const re of interdits) {
    const m = texte.match(new RegExp('.{0,40}' + re.source + '.{0,40}', re.flags));
    exiger(!m, 'Mot ou signe interdit dans le dossier (' + re.source + ') : « ' + (m ? m[0] : '') + ' »');
  }
  if (echecs.length) ecrire('## Échecs de cette génération', '', ...echecs.map((e) => '- ' + e), '');
  fs.mkdirSync(path.dirname(sortie), { recursive: true });
  fs.writeFileSync(sortie, L.join('\n'));
  console.log('Dossier écrit : ' + sortie + ' (' + L.length + ' lignes).');
  console.log('Postes : ' + tenus + ' tenus sur ' + total + ' ; règles : ' + CTRL.regles.filter((r) => r.ok).length + '/' + CTRL.regles.length + ' ; proposition identique au calcul complet : ' + identique + '.');
  console.log('Erreurs volontaires signalées par la bonne règle : ' + attrapees + '/' + nbErreurs + ' ; règles déclenchées : ' + couvertes + '/' + CTRL.regles.length + '.');
  console.log('Page dans Chrome : ' + (C ? (pageOk && pageModifOk ? 'conforme au moteur, et recalculée quand les données changent' : 'ÉCART') : 'non vérifiée (' + PAGEV.raison + ')') + '.');
  for (const a of avertissements) console.log('Avertissement : ' + a);
  if (echecs.length) { console.error(echecs.length + ' preuve(s) en échec :\n' + echecs.map((e) => ' - ' + e).join('\n')); process.exit(1); }
  console.log('Toutes les preuves du dossier passent.');
})().catch((e) => { console.error(e && e.stack ? e.stack : e); process.exit(1); });
