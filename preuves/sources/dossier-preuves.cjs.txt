ecrire('## 2. Le calcul des ' + total + ' postes', '',
  '**Un poste, c’est une personne à placer à un endroit, pendant une plage horaire, un jour donné.** Exemple : « piste 1000 m, commissaire de piste, lundi 2 juin, 10 h à 19 h ». Si deux commissaires de piste sont nécessaires, cela fait deux postes. Une personne tient au plus un poste par jour. ' + total + ' postes, c’est donc ' + total + ' journées (ou soirées) de travail à pourvoir dans la semaine.', '',
  '### 2.1 Le besoin de chaque activité (hypothèse de la maquette)', '',
  '| Activité | Poste | Métier exigé | En semaine | Le week-end | Exigence de niveau |', '|---|---|---|---|---|---|');
for (const p of d.postes) {
  const a = d.activites.find((x) => x.id === p.activite);

  contrat: 'Ce que disent le contrat de chaque CDI, ses congés et arrêts, et la réponse de chaque extra.',

  const indispo = d.sollicites.filter((id) => d.reponses[id] && !d.reponses[id][j.cle]).map((id) => personne(id).nom);

Indisponibles : ' + (indispo.join(', ') || 'aucun') + '. Sans réponse : ' + (sansReponse.join(', ') || 'aucun') + '.', '',

  for (const a of affs.slice().sort((x, y) => d.lignes.findIndex((l) => l.postes.includes(x.place.poste)) - d.lignes.findIndex((l) => l.postes.includes(y.place.poste)))) {

    ecrire('| ' + libelleLigne(a.place.poste) + (activiteDe(a.place.poste) && ['soiree', 'groupe', 'course'].includes(activiteDe(a.place.poste).id) ? ' (place d’expert)' : '') + ' | ' + p.nom + ' | ' + p.type + ' | ' + d.metiers[a.place.metier] + ' ' + p.metiers[a.place.metier] + ' |');

// ---------- 6. Preuve par personne ----------
ecrire('## 6. Preuve par personne', '', 'Pour chacun des ' + d.personnes.length + ' membres des viviers : ses affectations, et la vérification de chaque règle qui le concerne.', '',
  '| Personne | Statut | Métiers et niveaux | Absence connue | Ce qu’il a déclaré ou son contrat | Affectations | Jours | Heures | Repos le plus court | Repos le plus long | Disponibilité | Métier et niveau | Absences | Contrat ou pondération | Repos ≥ 11 h | Repos hebdo ≥ 35 h | ≤ 10 h par jour | ≤ 48 h et ≤ 6 jours |',
  '|---|---|---|---|---|---|---|---|---|---|---|---|---|---|---|---|---|---|');
for (const p of d.personnes) {
  const missions = [];
  d.jours.forEach((j, i) => { for (const a of PLAN.affectations[j.cle]) if (a.personne === p.id) missions.push({ i, j, a }); });
  const heures = missions.reduce((n, m) => n + dureeTravail(m.a.place), 0);
  let reposMin = null;
  for (let k = 1; k < missions.length; k++) {
    const e = missions[k].i * 1440 + missions[k].a.place.debut - (missions[k - 1].i * 1440 + missions[k - 1].a.place.fin);
    reposMin = reposMin == null ? e : Math.min(reposMin, e);
  }
  const abs = (p.absences || []).map((a) => a.motif + ' (' + a.jours.map((x) => jour(x).court).join(' ') + ')').join(', ') || 'aucune';
  let declare, dispoOk = true, contratOk = true;
  if (p.type === 'CDI') {
    const repos = d.jours.filter((j) => !(d.joursCDI[p.id] || []).includes(j.cle) && !(p.absences || []).some((a) => a.jours.includes(j.cle))).map((j) => j.court);
    declare = 'Contrat de 5 jours ; repos : ' + (repos.join(' ') || 'aucun (absence)');
    const presents = d.jours.filter((j) => !(p.absences || []).some((a) => a.jours.includes(j.cle))).length;
    contratOk = missions.length === Math.min(p.contrat.jours, presents);
  } else {
    const r = d.reponses[p.id];
    if (!d.sollicites.includes(p.id)) declare = 'Pas sollicité cette semaine';
    else if (!r) declare = 'Sollicité, pas de réponse';
    else declare = 'A répondu : ' + Object.entries(r).map(([k, v]) => jour(k).court + ' ' + (v === 'journee' ? 'journée' : v === 'soir' ? 'soirée' : 'journée et soirée')).join(', ');
    dispoOk = missions.every((m) => r && d.sollicites.includes(p.id) && (r[m.j.cle] === 'journee+soir' || (m.a.place.soir ? r[m.j.cle] === 'soir' : r[m.j.cle] === 'journee')));
    contratOk = missions.length <= P.PLAFOND[p.ponderation];
  }
  const metierOk = missions.every((m) => !!p.metiers[m.a.place.metier] && (!m.a.place.expert || p.metiers[m.a.place.metier] === 'Expert'));
  const absOk = missions.every((m) => !(p.absences || []).some((a) => a.jours.includes(m.j.cle)));
  const reposOk = reposMin == null || reposMin >= 11 * 60;
  const jourOk = missions.every((m) => dureeTravail(m.a.place) <= 10 * 60);
  const abs7 = missions.map((m) => [m.i * 1440 + m.a.place.debut, m.i * 1440 + m.a.place.fin]);
  let plusLong = abs7.length ? abs7[0][0] : 7 * 1440;
  for (let k = 1; k < abs7.length; k++) plusLong = Math.max(plusLong, abs7[k][0] - abs7[k - 1][1]);
  if (abs7.length) plusLong = Math.max(plusLong, 7 * 1440 - abs7[abs7.length - 1][1]);
  const hebdoOk = !abs7.length || plusLong >= 35 * 60;
  const semaineOk = heures <= 48 * 60 && missions.length <= 6;
  ecrire('| ' + p.nom + ' | ' + p.type + (p.ponderation ? ', pondération ' + p.ponderation : '') + ' | ' + metiers(p) + ' | ' + abs + ' | ' + declare + ' | ' +
    (missions.map((m) => m.j.court + ' ' + libelleLigne(m.a.place.poste).toLowerCase() + ', ' + h(m.a.place.debut) + ' à ' + h(m.a.place.fin)).join(' ; ') || 'aucune') + ' | ' + missions.length + ' | ' + enHeures(heures) + ' | ' +
    (reposMin == null ? 'sans objet' : enHeures(reposMin)) + ' | ' + (abs7.length ? enHeures(plusLong) : 'sans objet') + ' | ' + oui(dispoOk) + ' | ' + oui(metierOk) + ' | ' + oui(absOk) + ' | ' + oui(contratOk) + ' | ' + oui(reposOk) + ' | ' + oui(hebdoOk) + ' | ' + oui(jourOk) + ' | ' + oui(semaineOk) + ' |');
}
ecrire('', 'Les heures comptent le temps de présence, moins une heure de pause pour toute journée de plus de 6 heures.', '');

// ---------- 7. Tests ----------
ecrire('## 7. Tests du moteur', '', '### 7.1 Le moteur recalcule vraiment : d’autres entrées donnent un autre planning', '',
  '| Entrée | Jours des CDI | Postes tenus | Postes à pourvoir | Règles respectées | Ce qui a changé |', '|---|---|---|---|---|---|');
const comparer = (a, b) => d.jours.reduce((n, j) => n + a.affectations[j.cle].filter((x, k) => !b.affectations[j.cle][k] || b.affectations[j.cle][k].personne !== x.personne).length, 0);
const essais = [
  ['Sélection par défaut (10 extras, 9 réponses)', {}, d.sollicites],
  ['Sans Inès M.', { retires: ['ines'] }],
  ['Sans Inès M., Sarah C. et Noa B.', { retires: ['ines', 'sarah', 'noa'] }],
  ['Avec Chloé S. en plus (non sollicitée par défaut, réponse ajoutée pour le test)', { ajout: 'chloe' }],
];
for (const [nom, o] of essais) {
  let sol = d.sollicites.slice(), rep = Object.assign({}, d.reponses);
  if (o.retires) sol = sol.filter((x) => !o.retires.includes(x));
  if (o.ajout) { sol.push(o.ajout); rep[o.ajout] = { lun: 'journee', mar: 'journee', mer: 'journee' }; }
  for (const mode of ['figés', 'recalculés']) {
    if (!o.retires && !o.ajout && mode === 'recalculés') continue;
    const pl = mode === 'figés' ? P.proposer(d, { joursCDI: d.joursCDI, sollicites: sol, reponses: rep }) : P.proposer(d, { sollicites: sol, reponses: rep, essaisCDI: 40, joursCDIDepart: d.joursCDI });
    const c = P.controler(d, pl, { sollicites: sol, reponses: rep });
    const autres = c.regles.filter((r) => !r.regle.startsWith('Tous les postes'));
