/* Moteur de la maquette RKC.
   Prépare une proposition de planning à partir du programme, des besoins, des viviers et des disponibilités,
   la contrôle règle par règle, et cherche des remplaçants en cas d'imprévu.
   Script classique, sans module ni dépendance : il fonctionne tel quel dans la page et dans Node. */
(function (racine) {
  'use strict';

  const RANG = { 'Débutant': 1, 'Confirmé': 2, 'Expert': 3 };
  const POIDS = { forte: 3, normale: 2, faible: 1 };
  const PLAFOND = { forte: 5, normale: 4, faible: 1 };   // journées au plus dans la semaine, par pondération
  const REPOS_MIN = 11 * 60;                              // minutes entre la fin d'une mission et la suivante
  const JOURNEE_MAX = 10 * 60;                            // minutes de travail au plus dans une journée
  const HEBDO_MAX = 48 * 60;                              // minutes de travail au plus dans la semaine
  const ENCHAINEMENT_MAX = 60;                            // au plus une heure entre la fin de la journée et l'événement du soir
  const SEUL_SOIR = 300;                                  // faire venir un extra pour la seule soirée : s'il n'y a pas d'autre solution
  // La pause comprise dans une amplitude : 1 h pour 9 heures ou plus, 30 min au-delà de 6 heures (pratique des plannings du RKC)
  const pauseDe = (amplitude) => (amplitude >= 9 * 60 ? 60 : amplitude > 6 * 60 ? 30 : 0);
  const LIMITE_ESSAIS = 20000;

  const minutes = (h) => { const [a, b] = h.split(':').map(Number); return a * 60 + b; };
  const duree = (m) => Math.floor(m / 60) + ' h' + (m % 60 ? ' ' + String(m % 60).padStart(2, '0') : '');
  function texteHeure(m) {
    const x = ((m % 1440) + 1440) % 1440;
    const h = Math.floor(x / 60), mn = x % 60;
    return h + ' h' + (mn ? ' ' + String(mn).padStart(2, '0') : '');
  }

  function contexte(d) {
    return {
      d,
      jours: d.jours.map((j) => j.cle),
      jour: Object.fromEntries(d.jours.map((j) => [j.cle, j])),
      activite: Object.fromEntries(d.activites.map((a) => [a.id, a])),
      poste: Object.fromEntries(d.postes.map((p) => [p.id, p])),
      personne: Object.fromEntries(d.personnes.map((p) => [p.id, p])),
    };
  }

  const joursActivite = (ctx, a) => (a.jours === 'tous' ? ctx.jours : a.jours);

  // Horaires d'une activité un jour donné : ceux du jour, sinon ceux de la semaine ou du week-end, sinon les horaires communs
  function horairesActivite(ctx, a, jour) {
    const h = a.horaires || {};
    const x = h[jour] || (ctx.jour[jour] && ctx.jour[jour].weekend ? h.weekend : h.semaine) || [a.debut, a.fin];
    const debut = minutes(x[0]);
    let fin = minutes(x[1]);
    if (fin <= debut) fin += 1440;
    return { debut, fin };
  }
  // Horaires d'un poste : les siens s'il en a (par jour, semaine ou week-end), sinon ceux de son activité
  function horaires(ctx, poste, jour) {
    const a = ctx.activite[poste.activite];
    const propre = poste.horaires && (poste.horaires[jour] || (ctx.jour[jour] && ctx.jour[jour].weekend ? poste.horaires.weekend : poste.horaires.semaine));
    if (propre) {
      const debut = minutes(propre[0]);
      let fin = minutes(propre[1]);
      if (fin <= debut) fin += 1440;
      return { debut, fin };
    }
    if (poste.debut) {
      const debut = minutes(poste.debut);
      let fin = minutes(poste.fin);
      if (fin <= debut) fin += 1440;
      return { debut, fin };
    }
    return horairesActivite(ctx, a, jour);
  }

  function nombre(ctx, poste, jour) {
    if (typeof poste.nombre === 'number') return poste.nombre;
    if (poste.nombre[jour] != null) return poste.nombre[jour];
    return ctx.jour[jour].weekend ? poste.nombre.weekend : poste.nombre.semaine;
  }

  // Les places à tenir un jour donné. « ajouts » sert aux besoins nouveaux (un événement réservé au dernier moment).
  function placesDuJour(ctx, jour, ajouts) {
    const res = [];
    for (const p of ctx.d.postes.concat(ajouts || [])) {
      const a = ctx.activite[p.activite];
      if (!a || !joursActivite(ctx, a).includes(jour)) continue;
      if (p.jours && !p.jours.includes(jour)) continue;
      const n = nombre(ctx, p, jour);
      if (!n) continue;
      const { debut, fin } = horaires(ctx, p, jour);
      // Pause comprise dans l'amplitude : celle du poste, sinon une heure pour une journée de 9 heures ou plus,
      // 30 minutes au-delà de 6 heures (pratique des plannings du RKC : 1 h pour les journées, 30 min sinon)
      const amplitude = fin - debut;
      const pause = p.pause != null ? p.pause : pauseDe(amplitude);
      for (let i = 0; i < n; i++) {
        const expert = i < (p.experts || 0);
        res.push({
          id: jour + '/' + p.id + '/' + i, jour, poste: p.id, activite: p.activite, metier: p.metier,
          role: expert && p.roleExpert ? p.roleExpert : p.role,
          expert, niveauMin: p.niveauMin || null,
          encadre: (p.experts || 0) >= 1 && n >= 2 && i === n - 1, debut, fin, pause, soir: debut >= 17 * 60,
        });
      }
    }
    return res;
  }

  const absence = (p, jour) => (p.absences || []).find((a) => a.jours.includes(jour)) || null;

  // Un événement : un groupe, une privatisation, une course. La location, la Piste Découverte, les licenciés
  // et le magasin font la journée.
  const estEvenement = (ctx, id) => { const a = ctx.activite[id]; return !!a && (a.type === 'groupe' || a.type === 'course'); };

  // La journée d'une personne, un jour donné : de la première prise de poste à la dernière fin de poste des activités
  // du jour (hors événements), pour les rôles qu'elle tient. L'accueil a les horaires de l'accueil, un pistard ceux de
  // la location, le magasin les siens. Sans personne : la journée du site.
  function journee(ctx, jour, p) {
    const cle = jour + '|' + (p ? Object.keys(p.metiers).sort().join(',') : '');
    const memo = ctx.journees || (ctx.journees = {});
    if (cle in memo) return memo[cle];
    let de = Infinity, fin = -Infinity;
    for (const pl of placesDuJour(ctx, jour)) {
      if (estEvenement(ctx, pl.activite) || (p && !p.metiers[pl.metier])) continue;
      de = Math.min(de, pl.debut); fin = Math.max(fin, pl.fin);
    }
    return (memo[cle] = de < fin ? [de, fin] : null);
  }
  // Les événements du jour où une personne peut tenir un poste : ceux que l'appli lui montre
  function evenementsPour(ctx, jour, p) {
    const ids = [];
    for (const pl of placesDuJour(ctx, jour)) if (estEvenement(ctx, pl.activite) && p.metiers[pl.metier] && !ids.includes(pl.activite)) ids.push(pl.activite);
    return ids.sort((x, y) => horairesActivite(ctx, ctx.activite[x], jour).debut - horairesActivite(ctx, ctx.activite[y], jour).debut);
  }

  // ---------- La journée, puis la soirée : les mêmes personnes ----------
  // Le soir, le RKC prend d'abord ceux qui sont déjà là dans la journée : on ne fait pas venir un extra pour deux heures,
  // et quelqu'un présent depuis le matin sera là le soir. Deux places du même jour s'enchaînent quand la première est
  // une place de la journée et la seconde un événement qui commence à sa fin, ou au plus une heure après.
  function enchainables(ctx, a, b) {
    return !estEvenement(ctx, a.activite) && estEvenement(ctx, b.activite) && b.debut >= a.fin && b.debut - a.fin <= ENCHAINEMENT_MAX;
  }
  // Le travail d'une présence : une place seule, ou la journée et la soirée enchaînées, de la prise de poste à la fin
  // de la soirée, moins une seule pause (l'heure creuse entre les deux en tient lieu)
  function travailPresence(places) {
    if (places.length === 1) return places[0].fin - places[0].debut - places[0].pause;
    const t = places.slice().sort((x, y) => x.debut - y.debut);
    const amplitude = t[t.length - 1].fin - t[0].debut;
    let creux = 0;
    for (let k = 1; k < t.length; k++) creux += Math.max(0, t[k].debut - t[k - 1].fin);
    return amplitude - Math.max(pauseDe(amplitude), creux);
  }
  // Les places d'une personne un jour donné, regroupées en présences : la journée et la soirée enchaînées n'en font qu'une
  function blocsDuJour(ctx, places) {
    const t = places.slice().sort((x, y) => x.debut - y.debut);
    return t.length === 2 && enchainables(ctx, t[0], t[1]) ? [t] : t.map((x) => [x]);
  }
  const travailJour = (ctx, places) => blocsDuJour(ctx, places).reduce((n, b) => n + travailPresence(b), 0);
  // Les présences d'une personne dans un plan : [début, fin, travail, places], en minutes depuis le lundi 0 h
  function presencesDe(ctx, plan, id) {
    const res = [];
    ctx.jours.forEach((j, i) => {
      const pls = (plan.affectations[j] || []).filter((a) => a.personne === id).map((a) => a.place);
      for (const b of blocsDuJour(ctx, pls)) res.push([i * 1440 + b[0].debut, i * 1440 + b[b.length - 1].fin, travailPresence(b), b]);
    });
    return res;
  }

  // Les disponibilités, jour par jour, telles que l'extra les coche dans l'appli : « journee » pour la journée,
  // aux horaires de son rôle, et l'identifiant de chaque événement coché (on reprend ses horaires de ce jour-là).
  // Le moteur lit aussi « toute » (tout le jour et la nuit) et un créneau « 12:00-20:00 » (qui peut finir après minuit).
  function creneau(ctx, jour, s, p) {
    if (s === 'toute') return [0, 2880];
    if (s === 'journee') return journee(ctx, jour, p);
    if (ctx.activite[s]) { const h = horairesActivite(ctx, ctx.activite[s], jour); return [h.debut, h.fin]; }
    const [a, b] = s.split('-');
    const de = minutes(a);
    let fin = minutes(b);
    if (fin <= de) fin += 1440;
    return [de, fin];
  }
  // Les créneaux qui se touchent ou se chevauchent n'en font qu'un (l'événement du soir à la suite de la journée)
  function creneaux(ctx, jour, dispo, p) {
    if (!dispo) return [];
    const liste = (dispo === 'toute' ? ['toute'] : dispo).map((s) => creneau(ctx, jour, s, p)).filter(Boolean).sort((x, y) => x[0] - y[0]);
    const res = [];
    for (const c of liste) {
      const der = res[res.length - 1];
      if (der && c[0] <= der[1]) der[1] = Math.max(der[1], c[1]); else res.push(c.slice());
    }
    return res;
  }
  // Ce qui vaut pour une place : la journée pour un poste de la journée, l'événement lui-même pour un poste d'événement.
  // Cocher la journée ne dit rien de la soirée, ni cocher la soirée de la journée. « toute » et un créneau « de-à » valent
  // pour ce qu'ils couvrent.
  function couvre(ctx, jour, dispo, place, p) {
    if (!dispo) return false;
    const evt = estEvenement(ctx, place.activite);
    const vaut = (s) => (s === 'toute' ? true : s === 'journee' ? !evt : ctx.activite[s] ? (estEvenement(ctx, s) ? s === place.activite : !evt) : true);
    return creneaux(ctx, jour, (dispo === 'toute' ? ['toute'] : dispo).filter(vaut), p).some(([de, fin]) => de <= place.debut && place.fin <= fin);
  }
  // Disponible avant une heure donnée ce jour-là (pour la journée : avant 17 h ; pour le matin : avant midi)
  const dispoAvant = (ctx, jour, dispo, limite, p) => creneaux(ctx, jour, dispo, p).some(([de]) => de < limite);

  // A-t-elle le métier et le niveau pour cette place ? Un débutant ne tient qu'une place encadrée par un expert.
  function apte(p, place) {
    const niv = p.metiers[place.metier];
    if (!niv) return false;
    const r = RANG[niv];
    if (place.expert) return r === 3;
    if (place.niveauMin && r < RANG[place.niveauMin]) return false;
    if (r === 1 && !place.encadre) return false;
    return true;
  }

  // ---------- 1. Les jours des CDI : le contrat d'abord ----------

  function optionsCDI(ctx, p) {
    const presents = ctx.jours.filter((j) => !absence(p, j));
    const n = Math.min(p.contrat.jours, presents.length);
    if (presents.length <= n) return [presents];
    const k = presents.length - n;                       // jours de repos, pris d'affilée
    const res = [];
    for (let s = 0; s + k <= ctx.jours.length; s++) {
      const repos = ctx.jours.slice(s, s + k);
      if (repos.every((j) => presents.includes(j))) res.push({ jours: presents.filter((j) => !repos.includes(j)), weekend: repos.some((j) => ctx.jour[j].weekend) });
    }
    // Repos en semaine d'abord, quand l'activité est la plus faible ; le week-end seulement s'il n'y a pas d'autre choix
    const semaine = res.filter((o) => !o.weekend);
    const retenus = (semaine.length ? semaine : res).map((o) => o.jours);
    return retenus.length ? retenus : [presents.slice(0, n)];
  }

  function besoinsParJour(ctx) {
    const res = {};
    for (const j of ctx.jours) {
      const pls = placesDuJour(ctx, j);
      const parMetier = {}, experts = {};
      for (const pl of pls) {
        if (pl.soir) continue;
        parMetier[pl.metier] = (parMetier[pl.metier] || 0) + 1;
        if (pl.expert) experts[pl.metier] = (experts[pl.metier] || 0) + 1;
      }
      res[j] = { total: pls.length, journee: pls.filter((p) => !p.soir).length, parMetier, experts };
    }
    return res;
  }

  function noteCDI(ctx, combo, cdis, extras, reponses, besoins) {
    let note = 0;
    const dispoJour = (e, j) => !!reponses[e.id] && dispoAvant(ctx, j, reponses[e.id][j], 17 * 60, e);
    for (const j of ctx.jours) {
      const b = besoins[j];
      const presents = cdis.filter((p) => combo[p.id].includes(j));
      const offre = extras.filter((e) => dispoJour(e, j));
      if (presents.length > b.journee) note += 10000 * (presents.length - b.journee);   // CDI payé à attendre
      const manque = b.journee - presents.length - offre.length;
      if (manque > 0) note += 3000 * manque;
      for (const [m, n] of Object.entries(b.parMetier)) {
        const qui = presents.filter((p) => p.metiers[m]).length + offre.filter((e) => e.metiers[m]).length;
        if (qui < n) note += 2000 * (n - qui);
      }
      for (const [m, n] of Object.entries(b.experts)) {
        const qui = presents.filter((p) => p.metiers[m] === 'Expert').length + offre.filter((e) => e.metiers[m] === 'Expert').length;
        if (qui < n) note += 2000 * (n - qui);
      }
      note += (Math.max(0, b.journee - presents.length) * 10) / Math.max(1, offre.length);
    }
    for (const p of cdis) {
      for (const j of ctx.jours) {
        if (!combo[p.id].includes(j) && !absence(p, j) && ctx.jour[j].weekend) note += 400;   // repos le week-end : à éviter
      }
    }
    return note;
  }

  function combinaisonsCDI(ctx, cdis) {
    let combos = [{}];
    for (const p of cdis) {
      const opts = optionsCDI(ctx, p);
      const suivant = [];
      for (const c of combos) for (const o of opts) suivant.push(Object.assign({}, c, { [p.id]: o }));
      combos = suivant;
    }
    return combos;
  }

  // ---------- 2. Les places de chaque jour ----------

  function reposOk(etat, id, iJour, pl) {
    const f = etat.fin[id];
    return f == null || iJour * 1440 + pl.debut - f >= REPOS_MIN;
  }

  // Affectation optimale d'un jour : chaque place reçoit une personne, au moindre coût (méthode hongroise).
  function hongrois(cout) {
    const n = cout.length, m = cout[0].length;
    const u = new Array(n + 1).fill(0), v = new Array(m + 1).fill(0);
    const p = new Array(m + 1).fill(0), way = new Array(m + 1).fill(0);
    for (let i = 1; i <= n; i++) {
      p[0] = i;
      let j0 = 0;
      const minv = new Array(m + 1).fill(Infinity), used = new Array(m + 1).fill(false);
      do {
        used[j0] = true;
        const i0 = p[j0];
        let delta = Infinity, j1 = 0;
        for (let j = 1; j <= m; j++) {
          if (used[j]) continue;
          const cur = cout[i0 - 1][j - 1] - u[i0] - v[j];
          if (cur < minv[j]) { minv[j] = cur; way[j] = j0; }
          if (minv[j] < delta) { delta = minv[j]; j1 = j; }
        }
        for (let j = 0; j <= m; j++) {
          if (used[j]) { u[p[j]] += delta; v[j] -= delta; } else minv[j] -= delta;
        }
        j0 = j1;
      } while (p[j0] !== 0);
      do { const j1 = way[j0]; p[j0] = p[j1]; j0 = j1; } while (j0);
    }
    const res = new Array(n).fill(-1);
    for (let j = 1; j <= m; j++) if (p[j]) res[p[j] - 1] = j - 1;
    return res;
  }

  const INTERDIT = 1e7, VIDE = 1e5, CDI_PLACE = -1e4;

  // Un jour se résout en deux temps : les places de la journée, puis les événements du soir, tenus d'abord par ceux qui
  // sont déjà là et qui ont coché l'événement. Un extra ne vient pour la seule soirée que s'il n'y a pas d'autre solution.
  function resoudreJour(ctx, iJour, places, cdis, extras, reponses, etat, cdiTravailleDemain, extraDispoDemain) {
    const jour = ctx.jours[iJour];
    if (!places.length) return { affectations: [], vides: [], cout: 0 };
    const gens = cdis.concat(extras);
    const soirs = places.filter((pl) => pl.soir && estEvenement(ctx, pl.activite));
    const journee = places.filter((pl) => !soirs.includes(pl));
    const deja = (id) => etat.minutes[id] || 0;
    // Pourra-t-elle rester le soir après cette place ? Elle a coché l'événement, en a le rôle, et sa journée tient en 10 heures
    const resteLeSoir = (pl, p) => soirs.some((s) => apte(p, s) && enchainables(ctx, pl, s) && couvre(ctx, jour, reponses[p.id][jour], s, p)
      && travailPresence([pl, s]) <= JOURNEE_MAX && deja(p.id) + travailPresence([pl, s]) <= HEBDO_MAX);
    const prixJournee = (pl, p) => {
      if (!apte(p, pl) || !reposOk(etat, p.id, iJour, pl)) return INTERDIT;
      if (etat.interdits && etat.interdits.has(p.id + '|' + jour)) return INTERDIT;
      const niv = RANG[p.metiers[pl.metier]];
      // Garder les experts pour les places d'expert, et d'abord les experts CDI
      const gaspille = !pl.expert && niv === 3 ? (p.type === 'CDI' ? 8 : 5) : 0;
      if (p.type === 'CDI') {
        if (pl.soir) return INTERDIT;                              // les groupes du soir sont tenus par des extras
        const principal = Object.keys(p.metiers)[0] === pl.metier ? 0 : 60;
        return CDI_PLACE + principal + gaspille + (p.ordre || 0) * 0.001;
      }
      if (!couvre(ctx, jour, reponses[p.id][jour], pl, p)) return INTERDIT;
      if ((etat.joursExtra[p.id] || 0) >= PLAFOND[p.ponderation]) return INTERDIT;
      if (deja(p.id) + travailPresence([pl]) > HEBDO_MAX) return INTERDIT;
      const tour = 100 * (((etat.joursExtra[p.id] || 0) + 1) / POIDS[p.ponderation]);
      const soirPuisMatin = pl.soir && extraDispoDemain(p) ? 30 : 0;
      const debutant = niv === 1 ? 20 : 0;
      // Prioriser ceux qui sont disponibles pour tout : ils resteront pour l'événement du soir. Qui a coché la soirée mais ne
      // pourrait pas la tenir après cette place (journée trop longue) est gardé pour le soir, si d'autres peuvent tenir la journée.
      const soirPossible = soirs.length && soirs.some((s) => apte(p, s) && couvre(ctx, jour, reponses[p.id][jour], s, p));
      const pourTout = !soirPossible ? 0 : resteLeSoir(pl, p) ? -60 : 150;
      // Le directeur de course du soir : un expert qui a coché la soirée passe la journée sur place, puis dirige la soirée
      const dcDuSoir = pourTout < 0 && niv === 3 && soirs.some((s) => s.expert && apte(p, s) && enchainables(ctx, pl, s) && couvre(ctx, jour, reponses[p.id][jour], s, p) && travailPresence([pl, s]) <= JOURNEE_MAX) ? -120 : 0;
      return 200 + tour + gaspille + soirPuisMatin + debutant + pourTout + dcDuSoir + (p.ordre || 0) * 0.01;
    };
    const resoudre = (pls, prix) => {
      if (!pls.length) return [];
      const cout = pls.map((pl) => gens.map((p) => prix(pl, p)).concat(pls.map(() => VIDE)));
      const choix = hongrois(cout);
      const tenue = (k) => choix[k] >= 0 && choix[k] < gens.length && cout[k][choix[k]] < INTERDIT;
      // Un débutant n'est jamais seul : si la place d'expert de son poste reste vide, sa place l'est aussi
      for (let k = 0; k < pls.length; k++) {
        if (!tenue(k) || RANG[gens[choix[k]].metiers[pls[k].metier]] !== 1) continue;
        if (!pls.some((pl, x) => pl.poste === pls[k].poste && pl.expert && tenue(x))) choix[k] = -1;
      }
      return pls.map((pl, k) => (tenue(k) ? { place: pl, p: gens[choix[k]], cout: cout[k][choix[k]] } : { place: pl, p: null }));
    };
    const premier = resoudre(journee, prixJournee);
    const enJournee = {};
    for (const x of premier) if (x.p) enJournee[x.p.id] = x.place;
    const prixSoir = (pl, p) => {
      if (p.type === 'CDI' || !apte(p, pl)) return INTERDIT;      // les groupes du soir sont tenus par des extras
      if (etat.interdits && etat.interdits.has(p.id + '|' + jour)) return INTERDIT;
      if (!couvre(ctx, jour, reponses[p.id][jour], pl, p)) return INTERDIT;
      const niv = RANG[p.metiers[pl.metier]];
      const gaspille = !pl.expert && niv === 3 ? 5 : 0;
      const debutant = niv === 1 ? 20 : 0;
      const soirPuisMatin = extraDispoDemain(p) ? 30 : 0;
      const la = enJournee[p.id];
      if (la) {
        // Déjà là dans la journée : il enchaîne, si sa journée tient en 10 heures
        const t = travailPresence([la, pl]);
        if (!enchainables(ctx, la, pl) || t > JOURNEE_MAX || deja(p.id) + t > HEBDO_MAX) return INTERDIT;
        return 100 + gaspille + debutant + soirPuisMatin + (p.ordre || 0) * 0.01;
      }
      if (!reposOk(etat, p.id, iJour, pl)) return INTERDIT;
      if ((etat.joursExtra[p.id] || 0) >= PLAFOND[p.ponderation]) return INTERDIT;
      if (deja(p.id) + travailPresence([pl]) > HEBDO_MAX) return INTERDIT;
      const tour = 100 * (((etat.joursExtra[p.id] || 0) + 1) / POIDS[p.ponderation]);
      return 200 + SEUL_SOIR + tour + gaspille + debutant + soirPuisMatin + (p.ordre || 0) * 0.01;
    };
    const second = resoudre(soirs, prixSoir);
    const affectations = [], vides = [];
    let total = 0;
    for (const x of premier.concat(second).sort((a, b) => places.indexOf(a.place) - places.indexOf(b.place))) {
      if (x.p) { affectations.push({ place: x.place, personne: x.p.id }); total += x.cout - (x.p.type === 'CDI' ? CDI_PLACE : 0); }
      else vides.push(x.place);
    }
    return { affectations, vides, cout: total };
  }

  function affecterSemaine(ctx, joursCDI, extras, reponses, ajouts, interdits) {
    const etat = { joursExtra: {}, fin: {}, minutes: {}, interdits: interdits || new Set() };
    const affectations = {};
    const manques = [];
    let cout = 0;
    for (let i = 0; i < ctx.jours.length; i++) {
      const jour = ctx.jours[i];
      const demain = ctx.jours[i + 1];
      const cdis = ctx.d.personnes.filter((p) => p.type === 'CDI' && (joursCDI[p.id] || []).includes(jour));
      const ex = extras.filter((e) => reponses[e.id] && reponses[e.id][jour] && !absence(e, jour));
      const places = placesDuJour(ctx, jour, ajouts);
      const res = resoudreJour(
        ctx, i, places, cdis, ex, reponses, etat,
        (p) => !!demain && (joursCDI[p.id] || []).includes(demain),
        (p) => !!demain && !!reponses[p.id] && dispoAvant(ctx, demain, reponses[p.id][demain], 12 * 60, p),
      );
      affectations[jour] = res.affectations;
      manques.push(...res.vides);
      cout += res.cout;
      const parPersonne = {};
      for (const a of res.affectations) (parPersonne[a.personne] = parPersonne[a.personne] || []).push(a.place);
      for (const [id, pls] of Object.entries(parPersonne)) {
        etat.fin[id] = i * 1440 + Math.max(...pls.map((x) => x.fin));
        etat.minutes[id] = (etat.minutes[id] || 0) + travailJour(ctx, pls);
        if (ctx.personne[id].type === 'Extra') etat.joursExtra[id] = (etat.joursExtra[id] || 0) + 1;
      }
    }
    return { joursCDI, affectations, manques, cout };
  }

  // ---------- 2 bis. Le repos hebdomadaire : au moins 35 heures d'affilée dans la semaine ----------
  const REPOS_HEBDO = 35 * 60;
  function plusLongRepos(missions, finSemaine) {
    if (!missions.length) return finSemaine;
    const m = missions.slice().sort((x, y) => x[0] - y[0]);
    let plus = m[0][0];
    for (let k = 1; k < m.length; k++) plus = Math.max(plus, m[k][0] - m[k - 1][1]);
    return Math.max(plus, finSemaine - m[m.length - 1][1]);
  }
  function missionsDe(ctx, plan, id) {
    const res = [];
    ctx.jours.forEach((j, i) => { for (const a of plan.affectations[j] || []) if (a.personne === id) res.push([i * 1440 + a.place.debut, i * 1440 + a.place.fin]); });
    return res;
  }
  function reposHebdoKo(ctx, plan) {
    const fin = ctx.jours.length * 1440;
    return ctx.d.personnes.filter((p) => p.type === 'Extra').map((p) => p.id)
      .filter((id) => { const m = missionsDe(ctx, plan, id); return m.length && plusLongRepos(m, fin) < REPOS_HEBDO; });
  }
  // Si un extra n'a pas 35 heures d'affilée, on lui retire un jour et l'on recalcule, en gardant la meilleure solution
  function reparerReposHebdo(ctx, plan, recalcul) {
    let courant = plan, interdits = new Set();
    for (let tour = 0; tour < 12; tour++) {
      const ko = reposHebdoKo(ctx, courant);
      if (!ko.length) return courant;
      let mieux = null;
      for (const id of ko) {
        for (const j of ctx.jours.filter((x) => (courant.affectations[x] || []).some((a) => a.personne === id))) {
          const essai = new Set(interdits); essai.add(id + '|' + j);
          const res = recalcul(essai);
          if (res.manques.length > courant.manques.length) continue;
          const note = [reposHebdoKo(ctx, res).length, res.manques.length, res.cout];
          const meilleur = !mieux || note[0] < mieux.note[0] || (note[0] === mieux.note[0] && (note[1] < mieux.note[1] || (note[1] === mieux.note[1] && note[2] < mieux.note[2])));
          if (meilleur) mieux = { res, essai, note };
        }
      }
      if (!mieux || mieux.note[0] >= ko.length) return courant;
      courant = mieux.res; interdits = mieux.essai;
    }
    return courant;
  }

  // ---------- 3. La proposition complète ----------

  function proposer(d, opts) {
    opts = opts || {};
    const ctx = contexte(d);
    const reponses = opts.reponses || d.reponses;
    const sollicites = new Set(opts.sollicites || d.sollicites);
    const cdis = d.personnes.filter((p) => p.type === 'CDI');
    const extras = d.personnes.filter((p) => p.type === 'Extra' && sollicites.has(p.id) && reponses[p.id]);
    if (opts.joursCDI) {
      const base = affecterSemaine(ctx, opts.joursCDI, extras, reponses, opts.ajouts);
      return reparerReposHebdo(ctx, base, (interdits) => affecterSemaine(ctx, opts.joursCDI, extras, reponses, opts.ajouts, interdits));
    }
    const besoins = besoinsParJour(ctx);
    const combos = combinaisonsCDI(ctx, cdis)
      .map((c) => ({ c, note: noteCDI(ctx, c, cdis, extras, reponses, besoins) }))
      .sort((a, b) => a.note - b.note);
    if (opts.joursCDIDepart) combos.unshift({ c: opts.joursCDIDepart, note: -1 });
    // Repos du week-end, quand l'activité est la plus forte : à éviter pour les CDI
    const reposWeekend = (c) => cdis.reduce((n, p) => n + ctx.jours.filter((j) => ctx.jour[j].weekend && !absence(p, j) && !c[p.id].includes(j)).length, 0);
    let meilleur = null;
    for (const { c } of combos.slice(0, opts.essaisCDI || 60)) {
      const r = affecterSemaine(ctx, c, extras, reponses, opts.ajouts);
      r.cout += 150 * reposWeekend(c);
      if (!meilleur || r.manques.length < meilleur.manques.length ||
        (r.manques.length === meilleur.manques.length && r.cout < meilleur.cout)) meilleur = r;
    }
    return reparerReposHebdo(ctx, meilleur, (interdits) => affecterSemaine(ctx, meilleur.joursCDI, extras, reponses, opts.ajouts, interdits));
  }

  // ---------- 4. Le contrôle, règle par règle, sans faire confiance au moteur ----------

  function libellePoste(ctx, posteId) {
    const l = (ctx.d.lignes || []).find((x) => x.postes.includes(posteId));
    const p = ctx.poste[posteId] || {};
    const role = (p.role || '').toLowerCase();
    return (l ? l.libelle : posteId) + (role && l && l.libelle.toLowerCase() !== role ? ' (' + role + ')' : '');
  }

  // Quatre familles de règles, à ne pas confondre :
  //  - legal : minimums généraux du Code du travail (des dérogations encadrées existent par accord) ;
  //  - contrat : ce que disent le contrat et les données de chaque personne (congés, arrêt, réponses des extras) ;
  //  - rkc : règles d'organisation du RKC, paramétrables (hypothèses de la maquette) ;
  //  - ponderation : préférences et réglages d'optimisation.
  const FAMILLES = {
    legal: 'Règles légales générales',
    contrat: 'Contrats et données de chaque personne',
    rkc: 'Règles d’organisation du RKC, paramétrables',
    ponderation: 'Préférences et pondérations',
  };

  function controler(d, plan, opts) {
    opts = opts || {};
    const ctx = contexte(d);
    const reponses = opts.reponses || d.reponses;
    const sollicites = new Set(opts.sollicites || d.sollicites);
    const regles = [];
    const regle = (type, nom, erreurs, ref) => regles.push({ type, famille: FAMILLES[type], regle: nom, ref: ref || null, ok: erreurs.length === 0, erreurs });
    const toutes = [];
    for (let i = 0; i < ctx.jours.length; i++) {
      for (const a of plan.affectations[ctx.jours[i]] || []) toutes.push(Object.assign({ i }, a));
    }
    // Les présences de chacun : [début, fin, travail]. La journée et la soirée enchaînées n'en font qu'une, avec une seule pause.
    const parPersonne = {};
    for (const id of new Set(toutes.map((a) => a.personne))) parPersonne[id] = presencesDe(ctx, plan, id).sort((x, y) => x[0] - y[0]);
    let e;

    // ----- Règles légales générales -----
    e = [];
    for (const [id, missions] of Object.entries(parPersonne)) {
      for (let k = 1; k < missions.length; k++) {
        const ecart = missions[k][0] - missions[k - 1][1];
        if (ecart < 0) e.push(ctx.personne[id].nom + ' : deux postes qui se chevauchent');
        else if (ecart < REPOS_MIN) e.push(ctx.personne[id].nom + ' : ' + duree(ecart) + ' de repos seulement');
      }
    }
    regle('legal', 'Repos quotidien : au moins 11 heures entre la fin d’un poste et le début du suivant', e, 'Code du travail, art. L3131-1');

    e = [];
    const finSemaine = ctx.jours.length * 1440;
    for (const [id, missions] of Object.entries(parPersonne)) {
      let plusLong = missions[0][0];
      for (let k = 1; k < missions.length; k++) plusLong = Math.max(plusLong, missions[k][0] - missions[k - 1][1]);
      plusLong = Math.max(plusLong, finSemaine - missions[missions.length - 1][1]);
      if (plusLong < 35 * 60) e.push(ctx.personne[id].nom + ' : ' + duree(Math.max(0, plusLong)) + ' de repos d’affilée au plus dans la semaine');
    }
    regle('legal', 'Repos hebdomadaire : au moins 35 heures d’affilée dans la semaine, soit 24 heures plus les 11 heures quotidiennes', e, 'Code du travail, art. L3132-2');

    e = [];
    for (const [id, missions] of Object.entries(parPersonne)) {
      const parJour = {};
      for (const [deb, , t] of missions) { const j = Math.floor(deb / 1440); parJour[j] = (parJour[j] || 0) + t; }
      for (const [j, m] of Object.entries(parJour)) if (m > JOURNEE_MAX) e.push(ctx.personne[id].nom + ' : ' + duree(m) + ' de travail le ' + ctx.jour[ctx.jours[j]].nom.toLowerCase());
    }
    regle('legal', 'Durée quotidienne : pas plus de 10 heures de travail', e, 'Code du travail, art. L3121-18');

    e = [];
    for (const [id, missions] of Object.entries(parPersonne)) {
      const total = missions.reduce((n, m) => n + m[2], 0);
      if (total > 48 * 60) e.push(ctx.personne[id].nom + ' : ' + duree(total) + ' de travail dans la semaine');
    }
    regle('legal', 'Durée hebdomadaire : pas plus de 48 heures de travail', e, 'Code du travail, art. L3121-20');

    e = [];
    for (const [id, missions] of Object.entries(parPersonne)) {
      const n = new Set(missions.map(([deb]) => Math.floor(deb / 1440))).size;
      if (n > 6) e.push(ctx.personne[id].nom + ' : ' + n + ' jours de travail dans la semaine');
    }
    regle('legal', 'Pas plus de 6 jours de travail dans la semaine', e, 'Code du travail, art. L3132-1');

    // ----- Contrats et données de chaque personne -----
    e = [];
    for (const a of toutes) {
      const p = ctx.personne[a.personne];
      const abs = absence(p, ctx.jours[a.i]);
      if (abs) e.push(p.nom + ' planifié pendant : ' + abs.motif);
    }
    regle('contrat', 'Personne n’est planifié en congés, en arrêt maladie ou en déplacement', e);

    e = [];
    for (const a of toutes) {
      const p = ctx.personne[a.personne];
      if (p.type !== 'Extra') continue;
      if (!sollicites.has(p.id)) e.push(p.nom + ' planifié sans avoir été sollicité');
      else if (!reponses[p.id]) e.push(p.nom + ' planifié sans avoir répondu');
      else if (!couvre(ctx, ctx.jours[a.i], reponses[p.id][ctx.jours[a.i]], a.place, p)) e.push(p.nom + ' planifié sur ce qu’il n’a pas coché (' + ctx.jour[ctx.jours[a.i]].nom.toLowerCase() + ', ' + texteHeure(a.place.debut) + ' à ' + texteHeure(a.place.fin) + ')');
    }
    regle('contrat', 'Un extra ne travaille que sur ce qu’il a coché : la journée, aux horaires de son rôle, ou tel événement, à ses horaires', e);

    e = [];
    const travaillesCDI = {};
    for (const p of d.personnes.filter((x) => x.type === 'CDI')) {
      const presents = ctx.jours.filter((j) => !absence(p, j));
      const imprevus = ((opts.imprevus || {})[p.id] || []).length;   // absence survenue après la proposition
      const attendu = Math.min(p.contrat.jours, presents.length + imprevus) - imprevus;
      const travailles = ctx.jours.filter((j) => (plan.affectations[j] || []).some((a) => a.personne === p.id));
      travaillesCDI[p.id] = { presents, travailles };
      if (travailles.length !== attendu) e.push(p.nom + ' : ' + travailles.length + (travailles.length > 1 ? ' jours' : ' jour') + ' de travail au lieu des ' + attendu + ' prévus par son contrat');
    }
    regle('contrat', 'Chaque CDI travaille les jours prévus par son contrat (dans cette maquette, 5 par semaine l’été, 4 pour un temps partiel)', e);

    // ----- Règles d'organisation du RKC, paramétrables -----
    e = [];
    for (const j of ctx.jours) {
      const attendues = placesDuJour(ctx, j, opts.ajouts);
      const tenues = (plan.affectations[j] || []).map((a) => a.place.id);
      for (const pl of attendues) {
        const n = tenues.filter((x) => x === pl.id).length;
        if (n === 0) e.push(ctx.jour[j].nom + ' : ' + libellePoste(ctx, pl.poste) + ' sans personne');
        if (n > 1) e.push(ctx.jour[j].nom + ' : ' + libellePoste(ctx, pl.poste) + ' tenue deux fois');
      }
    }
    regle('rkc', 'Tous les postes sont tenus, sans trou', e);

    e = [];
    for (const j of ctx.jours) {
      const parP = {};
      for (const a of plan.affectations[j] || []) (parP[a.personne] = parP[a.personne] || []).push(a.place);
      for (const [id, pls] of Object.entries(parP)) {
        if (pls.length > 1 && blocsDuJour(ctx, pls).length > 1) e.push(ctx.jour[j].nom + ' : ' + ctx.personne[id].nom + ' sur ' + pls.length + ' postes qui ne s’enchaînent pas');
      }
    }
    regle('rkc', 'Une personne tient un seul poste à la fois : sa journée, puis au plus l’événement du soir qui la suit', e);

    e = [];
    for (const a of toutes) {
      const p = ctx.personne[a.personne];
      if (!apte(p, a.place)) e.push(ctx.jour[ctx.jours[a.i]].nom + ' : ' + p.nom + ' sans le métier ou le niveau pour ' + libellePoste(ctx, a.place.poste));
    }
    regle('rkc', 'Bon métier, bon niveau sur chaque poste', e);

    e = [];
    for (const j of ctx.jours) {
      const parPoste = {};
      for (const a of plan.affectations[j] || []) (parPoste[a.place.poste] = parPoste[a.place.poste] || []).push(a);
      for (const [poste, affs] of Object.entries(parPoste)) {
        const debutants = affs.filter((a) => RANG[ctx.personne[a.personne].metiers[a.place.metier]] === 1);
        const expert = affs.some((a) => a.place.expert && RANG[ctx.personne[a.personne].metiers[a.place.metier]] === 3);
        if (debutants.length > 1) e.push(ctx.jour[j].nom + ' : ' + debutants.length + ' débutants sur ' + libellePoste(ctx, poste));
        if (debutants.length && !expert) e.push(ctx.jour[j].nom + ' : débutant sans expert sur ' + libellePoste(ctx, poste));
      }
    }
    regle('rkc', 'Au plus un débutant par poste, toujours aux côtés d’un expert (pour les pistards : un commissaire de piste aux côtés d’un directeur de course)', e);

    e = [];
    for (const p of d.personnes.filter((x) => x.type === 'CDI')) {
      const { presents, travailles } = travaillesCDI[p.id];
      const malade = (opts.imprevus || {})[p.id] || [];
      const repos = presents.filter((j) => !travailles.includes(j) && !malade.includes(j));
      if (repos.length >= 2) {
        const idx = repos.map((j) => ctx.jours.indexOf(j));
        if (!idx.every((x, k) => k === 0 || x === idx[k - 1] + 1)) e.push(p.nom + ' : jours de repos non consécutifs');
      }
    }
    regle('rkc', 'Jours de repos des CDI pris d’affilée', e);

    // ----- Préférences et pondérations -----
    e = [];
    const joursExtra = {};
    for (const [id, pres] of Object.entries(parPersonne)) if (ctx.personne[id].type === 'Extra') joursExtra[id] = new Set(pres.map((m) => Math.floor(m[0] / 1440))).size;
    for (const [id, n] of Object.entries(joursExtra)) {
      const p = ctx.personne[id];
      if (n > PLAFOND[p.ponderation]) e.push(p.nom + ' : ' + n + ' jours de travail pour une pondération ' + p.ponderation + ' (' + PLAFOND[p.ponderation] + ' au plus)');
    }
    regle('ponderation', 'Pondération des extras : plafond de jours respecté (forte 5, normale 4, faible 1)', e);

    return { ok: regles.every((r) => r.ok), regles, places: toutes.length, manques: (plan.manques || []).length };
  }

  // ---------- 5. Les imprévus : qui appeler maintenant ? ----------

  function remplacants(d, plan, absentId, jour, opts) {
    opts = opts || {};
    const ctx = contexte(d);
    const reponses = opts.reponses || d.reponses;
    const sollicites = new Set(opts.sollicites || d.sollicites);
    const i = ctx.jours.indexOf(jour);
    const aff = (plan.affectations[jour] || []).find((a) => a.personne === absentId);
    const place = opts.place || (aff && aff.place);
    if (!place) return { place: null, candidats: [], ecartes: [] };
    const finVeille = {}, debutLendemain = {}, enPoste = {};
    if (i > 0) for (const a of plan.affectations[ctx.jours[i - 1]] || []) finVeille[a.personne] = Math.max(finVeille[a.personne] || 0, (i - 1) * 1440 + a.place.fin);
    if (i < ctx.jours.length - 1) for (const a of plan.affectations[ctx.jours[i + 1]] || []) debutLendemain[a.personne] = Math.min(debutLendemain[a.personne] == null ? Infinity : debutLendemain[a.personne], (i + 1) * 1440 + a.place.debut);
    for (const a of plan.affectations[jour] || []) (enPoste[a.personne] = enPoste[a.personne] || []).push(a.place);
    const deb = i * 1440 + place.debut, fin = i * 1440 + place.fin;
    const candidats = [], ecartes = [];
    for (const p of d.personnes) {
      if (p.id === absentId || !p.metiers[place.metier]) continue;
      const abs = absence(p, jour);
      if (abs) { ecartes.push({ id: p.id, nom: p.nom, raison: abs.motif }); continue; }
      const la = (enPoste[p.id] || []).filter((x) => x.id !== place.id);
      // Déjà là dans la journée : il peut enchaîner sur l'événement du soir, si sa journée tient en 10 heures
      const surPlace = la.length === 1 && p.type === 'Extra' && enchainables(ctx, la[0], place);
      if (la.length && !surPlace) { ecartes.push({ id: p.id, nom: p.nom, raison: 'déjà en poste ce jour-là', poste: la[0].poste }); continue; }
      if (surPlace && travailPresence([la[0], place]) > JOURNEE_MAX) { ecartes.push({ id: p.id, nom: p.nom, raison: 'déjà là dans la journée, mais sa journée dépasserait 10 heures', poste: la[0].poste }); continue; }
      if (!apte(p, place)) { ecartes.push({ id: p.id, nom: p.nom, raison: 'niveau insuffisant pour ce poste' }); continue; }
      if (p.type === 'CDI' && place.soir) { ecartes.push({ id: p.id, nom: p.nom, raison: 'les soirées sont tenues par des extras' }); continue; }
      const r = reponses[p.id];
      const aRepondu = p.type === 'Extra' && sollicites.has(p.id) && r;
      if (aRepondu && !couvre(ctx, jour, r[jour], place, p)) { ecartes.push({ id: p.id, nom: p.nom, raison: !r[jour] ? 'pas disponible ce jour-là' : estEvenement(ctx, place.activite) ? 'n’a pas coché cet événement' : 'n’a pas coché la journée' }); continue; }
      if (finVeille[p.id] != null && deb - finVeille[p.id] < REPOS_MIN) { ecartes.push({ id: p.id, nom: p.nom, raison: 'a fini à ' + texteHeure(finVeille[p.id]) + ' la veille, repos trop court' }); continue; }
      if (debutLendemain[p.id] != null && debutLendemain[p.id] - fin < REPOS_MIN) { ecartes.push({ id: p.id, nom: p.nom, raison: 'reprend trop tôt le lendemain' }); continue; }
      if (plusLongRepos(missionsDe(ctx, plan, p.id).concat([[deb, fin]]), ctx.jours.length * 1440) < REPOS_HEBDO) { ecartes.push({ id: p.id, nom: p.nom, raison: 'n’aurait plus 35 heures de repos d’affilée dans la semaine' }); continue; }
      if (presencesDe(ctx, ajouter(plan, jour, place, p.id), p.id).reduce((n, m) => n + m[2], 0) > HEBDO_MAX) { ecartes.push({ id: p.id, nom: p.nom, raison: 'dépasserait 48 heures de travail dans la semaine' }); continue; }
      if (p.type === 'CDI') {
        candidats.push({ id: p.id, nom: p.nom, type: 'CDI', rang: 2, statut: 'En repos ce jour-là', detail: 'seulement avec son accord' });
        continue;
      }
      const deja = ctx.jours.filter((j) => (plan.affectations[j] || []).some((a) => a.personne === p.id)).length;
      const semaine = deja ? 'déjà ' + deja + (deja > 1 ? ' jours' : ' jour') + ' de travail cette semaine' : '';
      if (!sollicites.has(p.id)) candidats.push({ id: p.id, nom: p.nom, type: 'Extra', rang: 3, statut: 'Pas sollicité cette semaine', detail: 'à appeler directement' });
      else if (!r) candidats.push({ id: p.id, nom: p.nom, type: 'Extra', rang: 3, statut: 'Pas de réponse à la demande', detail: 'à appeler directement' });
      else if (!surPlace && deja >= PLAFOND[p.ponderation]) candidats.push({ id: p.id, nom: p.nom, type: 'Extra', rang: 2.5, statut: 'Disponible, mais au-delà de sa pondération', detail: semaine });
      else candidats.push({ id: p.id, nom: p.nom, type: 'Extra', rang: 1, surPlace, statut: surPlace ? 'Déjà là dans la journée, a coché cet événement et enchaîne après son poste' : estEvenement(ctx, place.activite) ? 'A coché cet événement dans l’appli' : 'A coché la journée dans l’appli', detail: semaine, veille: finVeille[p.id] != null ? texteHeure(finVeille[p.id]) : null });
    }
    const poids = (c) => (ctx.personne[c.id].ponderation ? POIDS[ctx.personne[c.id].ponderation] : 0);
    const niveau = (c) => RANG[ctx.personne[c.id].metiers[place.metier]];
    // Ceux qui sont déjà là d'abord : on ne fait pas venir quelqu'un pour la seule soirée s'il y a une autre solution
    candidats.sort((a, b) => a.rang - b.rang || Number(!!b.surPlace) - Number(!!a.surPlace) || niveau(b) - niveau(a) || poids(b) - poids(a) || (ctx.personne[a.id].ordre || 0) - (ctx.personne[b.id].ordre || 0));
    return { place, candidats, ecartes };
  }

  function remplacer(plan, jour, absentId, remplacantId) {
    const copie = JSON.parse(JSON.stringify(plan));
    for (const a of copie.affectations[jour]) {
      if (a.personne === absentId) { a.personne = remplacantId; a.remplace = absentId; }
    }
    return copie;
  }

  function ajouter(plan, jour, place, id) {
    const copie = JSON.parse(JSON.stringify(plan));
    (copie.affectations[jour] = copie.affectations[jour] || []).push({ place, personne: id, ajout: true });
    return copie;
  }

  // Un événement ajouté après la validation : l'équipe qu'il demande, place par place, avec qui peut la tenir.
  // Les places d'expert d'abord ; chaque personne retenue entre dans le planning avant la recherche suivante.
  function renforts(d, plan, ajouts, jour, opts) {
    opts = opts || {};
    const ctx = contexte(d);
    const ids = new Set(ajouts.map((a) => a.id));
    const places = placesDuJour(ctx, jour, ajouts).filter((p) => ids.has(p.poste));
    places.sort((a, b) => (b.expert - a.expert) || (Number(a.encadre) - Number(b.encadre)));
    let courant = plan;
    const res = [];
    for (const place of places) {
      const r = remplacants(d, courant, '', jour, Object.assign({}, opts, { place }));
      const premier = r.candidats.find((c) => c.rang === 1) || null;
      if (premier) courant = ajouter(courant, jour, place, premier.id);
      res.push({ place, candidats: r.candidats, ecartes: r.ecartes, choisi: premier ? premier.id : null });
    }
    return { places: res, plan: courant };
  }

  // ---------- 6. La grille d'affichage : une ligne par endroit, une colonne par jour ----------

  function grille(d, plan) {
    const ctx = contexte(d);
    return d.lignes.map((l) => ({
      id: l.id,
      libelle: l.libelle,
      jours: ctx.jours.map((j) => {
        const affs = (plan.affectations[j] || []).filter((a) => l.postes.includes(a.place.poste));
        const vides = (plan.manques || []).filter((m) => m.jour === j && l.postes.includes(m.poste));
        return {
          jour: j,
          actif: affs.length + vides.length > 0,
          personnes: affs.map((a) => ({
            id: a.personne,
            nom: ctx.personne[a.personne].nom,
            type: ctx.personne[a.personne].type,
            poste: a.place.poste,
            role: a.place.role || (ctx.poste[a.place.poste] ? ctx.poste[a.place.poste].role : ''),
            expert: a.place.expert,
            debut: a.place.debut, fin: a.place.fin, pause: a.place.pause,
            horaire: texteHeure(a.place.debut) + ' à ' + texteHeure(a.place.fin),
            remplace: a.remplace ? ctx.personne[a.remplace].nom : null,
          })),
          vides: vides.length,
        };
      }),
    }));
  }

  racine.PlanningRKC = {
    proposer, controler, remplacants, remplacer, ajouter, renforts, grille, texteHeure,
    creneaux: (d, jour, dispo, personne) => { const ctx = contexte(d); return creneaux(ctx, jour, dispo, typeof personne === 'string' ? ctx.personne[personne] : personne); },
    journee: (d, jour, personne) => { const ctx = contexte(d); return journee(ctx, jour, typeof personne === 'string' ? ctx.personne[personne] : personne); },
    evenementsPour: (d, jour, personne) => { const ctx = contexte(d); return evenementsPour(ctx, jour, typeof personne === 'string' ? ctx.personne[personne] : personne); },
    estEvenement: (d, id) => estEvenement(contexte(d), id),
    presences: (d, plan, id) => presencesDe(contexte(d), plan, id),
    enchainables: (d, a, b) => enchainables(contexte(d), a, b),
    horairesActivite: (d, id, jour) => { const ctx = contexte(d); return horairesActivite(ctx, ctx.activite[id], jour); },
    libellePoste: (d, id) => libellePoste(contexte(d), id),
    placesDuJour: (d, jour, ajouts) => placesDuJour(contexte(d), jour, ajouts),
    RANG, POIDS, PLAFOND, REPOS_MIN, JOURNEE_MAX, HEBDO_MAX, ENCHAINEMENT_MAX, FAMILLES,
  };
})(typeof window !== 'undefined' ? window : globalThis);
